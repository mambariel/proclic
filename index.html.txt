        const msgEl = document.getElementById("confirmMessage");
        const cancelBtn = document.getElementById("confirmCancelBtn");

        successBtn.onclick = () => fechar(true);
      });
    }

      try {

        const dados = await resp.json();
        estado = {
          roadmap: dados.roadmap || [],
          assinados: dados.assinados || [],
          mapa_frentes: dados.mapa_frentes || []
        };
        esconderAviso();
        renderTudo();
      } catch (erro) {
        aviso("Problema ao conectar ao Google Sheets. Certifique-se de que o Apps Script está ativo e configurado como 'Qualquer pessoa'.");
        console.error(erro);
      }
    }

    function montarEtapas() {
      return etapasModelo.map(etapa => ({
        ...etapa,
        itens: estado.roadmap.filter(r => String(r.etapa) === String(etapa.id))
      }));
    }

    function renderSelects() {
      const filtro = document.getElementById("filtroEtapa");
      const cadastro = document.getElementById("etapaCadastro");
      filtro.innerHTML = `<option value="todas">Todas as etapas</option>` + etapasModelo.map(e => `<option value="${e.id}">${e.id} — ${e.titulo}</option>`).join("");
      cadastro.innerHTML = etapasModelo.map(e => `<option value="${e.id}">Etapa ${e.id}</option>`).join("");
    }

    function renderStats() {
      const etapas = montarEtapas();
      const total = estado.roadmap.length;
      const maior = etapas.reduce((a, b) => a.itens.length > b.itens.length ? a : b, etapas[0]);
      const qtdAvancadas = estado.roadmap.filter(r => ["2", "5", "6"].includes(String(r.etapa))).length;
      const progresso = Math.round((qtdAvancadas / Math.max(total, 1)) * 100);

      document.getElementById("stats").innerHTML = `
        <div class="stat"><strong>${total}</strong><small>entes/órgãos monitorados</small></div>
        <div class="stat"><strong>${maior.itens.length}</strong><small>em ${maior.titulo}</small></div>
        <div class="stat"><strong>6</strong><small>fases do fluxo</small></div>
        <div class="stat" style="background:rgba(16,185,129,.14)"><strong>${estado.assinados.length}</strong><small>parcerias assinadas</small></div>
      `;
      document.getElementById("progressBar").style.width = progresso + "%";
    }

    function renderRoadmap() {
      const busca = slug(document.getElementById("busca").value);
      const etapaAtiva = document.getElementById("filtroEtapa").value;
      const etapas = montarEtapas()
        .filter(e => etapaAtiva === "todas" || String(e.id) === etapaAtiva)
        .map(e => ({ ...e, itens: e.itens.filter(item => slug(item.nome).includes(busca)) }))
        .filter(e => e.itens.length > 0 || etapaAtiva !== "todas");

      document.getElementById("roadmap").innerHTML = etapas.map((etapa, index) => `
        <article class="stage" ondragover="permitirSoltar(event)" ondrop="soltarNaEtapa(event, ${etapa.id})">
          <div class="stage-header ${etapa.cor}">
            <div><small>Etapa ${etapa.id}</small><h3 class="font-bold text-white">${etapa.titulo}</h3></div>
            <span class="badge">${index <= 1 ? "Concluído" : index === 2 ? "Em andamento" : "Atenção"}</span>
          </div>
          ${etapa.itens.map(item => `
            <div class="partner" draggable="true" ondragstart="iniciarArraste(event, '${item.id}')">
              <span class="flex items-center gap-2">
                <i data-lucide="grip-vertical" class="w-4 h-4 opacity-70"></i>
                ${item.nome}
              </span>
              <span class="partner-actions">
                <button class="btn danger small flex items-center justify-center p-1" onclick="excluirRoadmap('${item.id}')" title="Excluir">
                  <i data-lucide="trash-2" class="w-3.5 h-3.5"></i>
                </button>
              </span>
            </div>
          `).join("")}
        </article>
      `).join("");
      atualizarIcones();
    }

    function iniciarArraste(event, id) {
      event.dataTransfer.setData("text/plain", id);
    }

    function permitirSoltar(event) {
      event.preventDefault();
    }

    async function soltarNaEtapa(event, etapaDestino) {
      event.preventDefault();
      const id = event.dataTransfer.getData("text/plain");
      const item = estado.roadmap.find(r => String(r.id) === String(id));
      if (!item) return;
      await apiPost({ acao: "editar", aba: "roadmap", id, registro: { ...item, etapa: String(etapaDestino) } });
      await carregarDados();
    }

    async function cadastrarParceiro(event) {
      event.preventDefault();
      const nome = document.getElementById("novoParceiro").value.trim();
      const etapa = document.getElementById("etapaCadastro").value;
      if (!nome) return;

      const existente = estado.roadmap.find(r => slug(r.nome) === slug(nome));
      if (existente) {
        await apiPost({ acao: "editar", aba: "roadmap", id: existente.id, registro: { ...existente, etapa } });
      } else {
        await apiPost({ acao: "criar", aba: "roadmap", registro: { id: uid(), nome, etapa } });
      }

      document.getElementById("novoParceiro").value = "";
      await carregarDados();
    }

    async function excluirRoadmap(id) {
      const confirmado = await mostrarConfirmacao("Confirmar Exclusão", "Tem certeza de que deseja remover este parceiro do Roadmap institucional?");
      if (!confirmado) return;
      await apiPost({ acao: "excluir", aba: "roadmap", id });
      await carregarDados();
    }

    function carregarFotoAssinado(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = () => {
        fotoAssinadoAtual = reader.result;
        document.getElementById("fotoLabel").textContent = "Foto selecionada";
        document.getElementById("fotoPreview").innerHTML = `<div class="preview"><img src="${fotoAssinadoAtual}" alt="Prévia"><div><strong>Prévia da imagem</strong><br><small>A imagem será salva no card da parceria.</small></div></div>`;
      };
      reader.readAsDataURL(file);
    }

    async function cadastrarAssinado(event) {
      event.preventDefault();
      const input = document.getElementById("novoAssinado");
      const nome = input.value.trim();
      if (!nome) return;

      const existente = estado.assinados.find(a => slug(a.nome) === slug(nome));
      if (existente) {
        await apiPost({ acao: "editar", aba: "assinados", id: existente.id, registro: { ...existente, nome, foto: fotoAssinadoAtual || existente.foto || "" } });
      } else {
        await apiPost({ acao: "criar", aba: "assinados", registro: { id: uid(), nome, foto: fotoAssinadoAtual } });
      }

      const noRoadmap = estado.roadmap.find(r => slug(r.nome) === slug(nome));
      if (noRoadmap) {
        await apiPost({ acao: "excluir", aba: "roadmap", id: noRoadmap.id });
      }

      input.value = "";
      fotoAssinadoAtual = "";
      document.getElementById("fotoLabel").textContent = "Adicionar foto/logo";
      document.getElementById("fotoPreview").innerHTML = "";
      await carregarDados();
    }

    async function excluirAssinado(id) {
      const confirmado = await mostrarConfirmacao("Confirmar Exclusão", "Deseja mesmo remover esta parceria formalizada dos registros?");
      if (!confirmado) return;
      await apiPost({ acao: "excluir", aba: "assinados", id });
      await carregarDados();
    }

    function renderAssinados() {
      document.getElementById("assinadosCount").textContent = `${estado.assinados.length} integrações formalizadas`;
      document.getElementById("assinados").innerHTML = estado.assinados.map(a => `
        <div class="signed-card">
          <div class="logo-box">${a.foto ? `<img src="${a.foto}" alt="${a.nome}">` : `<span class="text-xl font-black">${a.nome.slice(0, 2).toUpperCase()}</span>`}</div>
          <h3 class="font-bold text-white mt-3">${a.nome}</h3>
          <p class="text-xs text-white/60 mb-3">Convênio e integração ativa</p>
          <button class="btn danger small w-full flex items-center justify-center gap-1.5 py-2" onclick="excluirAssinado('${a.id}')">
            <i data-lucide="trash-2" class="w-3.5 h-3.5"></i>
            Excluir
          </button>
        </div>
      `).join("");
      atualizarIcones();
    }

    async function salvarMapaFrente(event) {
      event.preventDefault();
      const registro = {
        id: editandoMapaId || uid(),
        parceiro: document.getElementById("mapaParceiro").value.trim(),
        logo: document.getElementById("mapaLogo").value.trim(),
        cor: document.getElementById("mapaCor").value,
        frente: document.getElementById("mapaFrente").value.trim(),
        objetivo: document.getElementById("mapaObjetivo").value.trim(),
        etapa: document.getElementById("mapaEtapa").value.trim(),
        status: document.getElementById("mapaStatus").value.trim() || "Em andamento",
        acao: document.getElementById("mapaAcao").value.trim()
      };

      if (!registro.parceiro || !registro.frente) return;

      if (editandoMapaId) {
        await apiPost({ acao: "editar", aba: "mapa_frentes", id: editandoMapaId, registro });
      } else {
        await apiPost({ acao: "criar", aba: "mapa_frentes", registro });
      }

      limparFormularioMapa();
      await carregarDados();
    }

    function editarMapaFrente(id) {
      const f = estado.mapa_frentes.find(item => String(item.id) === String(id));
      if (!f) return;
      document.getElementById("mapaParceiro").value = f.parceiro || "";
      document.getElementById("mapaLogo").value = f.logo || "";
      document.getElementById("mapaCor").value = f.cor || "azul";
      document.getElementById("mapaFrente").value = f.frente || "";
      document.getElementById("mapaEtapa").value = f.etapa || "";
      document.getElementById("mapaStatus").value = f.status || "Em andamento";
      document.getElementById("mapaObjetivo").value = f.objetivo || "";
      document.getElementById("mapaAcao").value = f.acao || "";
      editandoMapaId = id;
      document.getElementById("tituloFormMapa").textContent = "Editar frente estratégica";
      document.getElementById("botaoSalvarMapa").textContent = "Salvar alterações";
      document.getElementById("cancelarEdicao").style.display = "inline-flex";
      window.scrollTo({ top: document.getElementById("tituloFormMapa").offsetTop - 120, behavior: "smooth" });
    }

    function limparFormularioMapa() {
      ["mapaParceiro", "mapaLogo", "mapaFrente", "mapaEtapa", "mapaObjetivo", "mapaAcao"].forEach(id => document.getElementById(id).value = "");
      document.getElementById("mapaStatus").value = "Em andamento";
      document.getElementById("mapaCor").value = "azul";
      editandoMapaId = null;
      document.getElementById("tituloFormMapa").textContent = "Cadastrar nova frente";
      document.getElementById("botaoSalvarMapa").textContent = "Cadastrar no mapa";
      document.getElementById("cancelarEdicao").style.display = "none";
    }

    async function excluirFrenteMapa(id) {
      const confirmado = await mostrarConfirmacao("Confirmar Exclusão", "Deseja mesmo remover esta frente operacional do painel de monitoramento?");
      if (!confirmado) return;
      await apiPost({ acao: "excluir", aba: "mapa_frentes", id });
      await carregarDados();
    }

    async function excluirParceiroMapa(parceiro) {
      const confirmado = await mostrarConfirmacao("Confirmar Exclusão do Parceiro", `Deseja mesmo excluir o parceiro "${parceiro}" e todas as frentes de trabalho vinculadas?`);
      if (!confirmado) return;
      const frentes = estado.mapa_frentes.filter(f => f.parceiro === parceiro);
      aviso("Excluindo todas as frentes vinculadas...");
      for (const frente of frentes) {
        await apiPost({ acao: "excluir", aba: "mapa_frentes", id: frente.id });
      }
      await carregarDados();
    }

    function corHeader(cor) {
      const cores = {
        azul: "linear-gradient(90deg, rgba(59,130,246,.25), rgba(99,102,241,.16))",
        vermelho: "linear-gradient(90deg, rgba(239,68,68,.25), rgba(0,0,0,.16))",
        verde: "linear-gradient(90deg, rgba(16,185,129,.25), rgba(6,182,212,.16))",
        ambar: "linear-gradient(90deg, rgba(245,158,11,.25), rgba(249,115,22,.16))"
      };
      return cores[cor] || cores.azul;
    }

                    </div>

                      <p class="text-white/80 text-sm mt-1 leading-relaxed">${f.objetivo || f.objective || "-"}</p>
                    </div>
                  </div>
                  <div>
                    <div class="my-2">
                      <span class="label">Etapa</span>
                      <div class="mt-1"><span class="stage-pill">${f.etapa || "-"}</span></div>
                    </div>
                    <div class="mt-2">
                      <span class="label">Ação / Pendência</span>

                    </div>
                  </div>
                </div>
              `).join("")}
            </div>
          </article>
        `;
      }).join("");
      atualizarIcones();

    function renderTudo() {
      renderSelects();
      renderStats();
      renderRoadmap();
      renderAssinados();
      renderMapaFrentes();
    }
